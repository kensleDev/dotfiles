import { writable, get } from 'svelte/store';
import type { Printer, PrinterModelType } from '@/printer';

interface StreamInfo {
    printerId: number;
    url: string;
    active: boolean;
    error: string | null;
    reconnectAttempts: number;
    lastAttempt: number;
}

interface CameraState {
    streams: Map<number, StreamInfo>;
    maxConcurrentStreams: number;
}

function createCameraProxyService() {
    const { subscribe, update } = writable<CameraState>({
        streams: new Map(),
        maxConcurrentStreams: 5
    });

    const streamsMap = new Map<number, StreamInfo>();
    const reconnectTimeouts = new Map<number, NodeJS.Timeout>();
    const proxyPorts = new Map<number, number>();

    const PROXY_START_PORT = 8081;

    // eslint-disable-next-line @typescript-eslint/no-unused-vars
    function getStreamType(_model: PrinterModelType): 'jpeg' | 'rtsp' {
        // For now, all models support JPEG streaming on port 6000
        // RTSP support can be added later if needed
        return 'jpeg';
    }

    function getNextProxyPort(): number {
        const usedPorts = new Set(proxyPorts.values());
        let port = PROXY_START_PORT;
        while (usedPorts.has(port)) {
            port++;
        }
        return port;
    }

    function getStreamUrl(printer: Printer): string {
        return `/api/printers/${printer.id}/camera`;
    }

    function startStream(printer: Printer): Promise<string> {
        return new Promise((resolve, reject) => {
            const state = get({ subscribe });

            if (state.streams.size >= state.maxConcurrentStreams) {
                reject(new Error('Maximum concurrent streams reached'));
                return;
            }

            const existingStream = state.streams.get(printer.id);
            if (existingStream && existingStream.active) {
                resolve(existingStream.url);
                return;
            }

            const streamUrl = getStreamUrl(printer);

            const streamInfo: StreamInfo = {
                printerId: printer.id,
                url: streamUrl,
                active: true,
                error: null,
                reconnectAttempts: 0,
                lastAttempt: Date.now()
            };

            streamsMap.set(printer.id, streamInfo);
            update((s) => ({
                ...s,
                streams: new Map(streamsMap)
            }));

            resolve(streamUrl);
        });
    }

    function stopStream(printerId: number): void {
        const timeout = reconnectTimeouts.get(printerId);
        if (timeout) {
            clearTimeout(timeout);
            reconnectTimeouts.delete(printerId);
        }

        streamsMap.delete(printerId);
        const port = proxyPorts.get(printerId);
        if (port) {
            proxyPorts.delete(printerId);
        }

        update((s) => ({
            ...s,
            streams: new Map(streamsMap)
        }));
    }

    function cleanupAll(): void {
        reconnectTimeouts.forEach((timeout) => clearTimeout(timeout));
        reconnectTimeouts.clear();
        streamsMap.clear();
        proxyPorts.clear();
        update((s) => ({
            ...s,
            streams: new Map()
        }));
    }

    function getStream(printerId: number): StreamInfo | undefined {
        return streamsMap.get(printerId);
    }

    function isStreamActive(printerId: number): boolean {
        return streamsMap.get(printerId)?.active ?? false;
    }

    function updateStreamError(printerId: number, error: string | null): void {
        const stream = streamsMap.get(printerId);
        if (stream) {
            stream.error = error;
            stream.active = error === null;
            streamsMap.set(printerId, stream);
            update((s) => ({
                ...s,
                streams: new Map(streamsMap)
            }));
        }
    }

    return {
        subscribe,
        startStream,
        stopStream,
        cleanupAll,
        getStreamUrl,
        getStream,
        isStreamActive,
        updateStreamError,
        getStreamType,
        getNextProxyPort
    };
}

export const cameraProxyService = createCameraProxyService();
